import React, { createContext, useContext, useEffect, useMemo, useState } from 'react';
import React, { createContext, useCallback, useContext, useEffect, useMemo, useState } from 'react';

type SignUpResult = {
  needsEmailConfirmation: boolean;
};

type UserRole = 'cliente' | 'dueno';

type AuthContextValue = {
  session: Session | null;
  user: User | null;
  loading: boolean;
  role: UserRole;
  isDueno: boolean;
  signIn: (email: string, password: string) => Promise<void>;
  signUp: (username: string, email: string, password: string) => Promise<SignUpResult>;
  signOut: () => Promise<void>;
  refreshRole: () => Promise<void>;
};

const AuthContext = createContext<AuthContextValue | undefined>(undefined);

export function AuthProvider({ children }: { children: React.ReactNode }) {
  const [session, setSession] = useState<Session | null>(null);
  const [loading, setLoading] = useState(true);
  const [role, setRole] = useState<UserRole>('cliente');

  const fetchRole = useCallback(async (userId: string) => {
    try {
      const { count, error } = await supabase
        .from('comercios')
        .select('id', { count: 'exact', head: true })
        .eq('user_id', userId);

      if (error) throw error;
      setRole((count ?? 0) > 0 ? 'dueno' : 'cliente');
    } catch {
      setRole('cliente');
    }
  }, []);

  const refreshRole = useCallback(async () => {
    if (session?.user) {
      await fetchRole(session.user.id);
    }
  }, [session?.user, fetchRole]);

  useEffect(() => {
    let isMounted = true;

    supabase.auth.getSession().then(({ data }) => {
      if (isMounted) {
        setSession(data.session);
        setLoading(false);
        if (data.session?.user) {
          fetchRole(data.session.user.id);
        }
      }
    });

    const { data: authListener } = supabase.auth.onAuthStateChange((_event, nextSession) => {
      setSession(nextSession);
      setLoading(false);
      if (nextSession?.user) {
        fetchRole(nextSession.user.id);
      } else {
        setRole('cliente');
      }
    });

    return () => {
      isMounted = false;
      authListener.subscription.unsubscribe();
    };
  }, []);
  }, [fetchRole]);

  const value = useMemo<AuthContextValue>(() => ({
    session,
    user: session?.user ?? null,
    loading,
    role,
    isDueno: role === 'dueno',
    signIn: async (email: string, password: string) => {
      const { error } = await supabase.auth.signInWithPassword({
        email: email.trim().toLowerCase(),
        password,
    signOut: async () => {
      const { error } = await supabase.auth.signOut();
      if (error) throw error;
    },
  }), [loading, session]);
    refreshRole,
  }), [loading, session, role, refreshRole]);

  return <AuthContext.Provider value={value}>{children}</AuthContext.Provider>;
}

-- Migracion: Agregar campos de control de publicacion a productos.
-- Ejecutar en el SQL Editor de Supabase.

BEGIN;

-- Agregar campo publicado (default true para productos existentes)
ALTER TABLE productos
  ADD COLUMN IF NOT EXISTS publicado BOOLEAN NOT NULL DEFAULT true;

-- Agregar campo disponible si no existe (default true)
ALTER TABLE productos
  ADD COLUMN IF NOT EXISTS disponible BOOLEAN NOT NULL DEFAULT true;

-- Agregar RLS a productos si no existe
ALTER TABLE productos ENABLE ROW LEVEL SECURITY;

-- Política para productos: todos los autenticados pueden ver, solo el dueño del comercio puede modificar
DROP POLICY IF EXISTS "Users can view all products" ON productos;
DROP POLICY IF EXISTS "Users can manage own commerce products" ON productos;

CREATE POLICY "Users can view all products" ON productos
  FOR SELECT
  TO authenticated
  USING (true);

CREATE POLICY "Users can manage own commerce products" ON productos
  FOR ALL
  TO authenticated
  USING (
    EXISTS (
      SELECT 1
      FROM comercios
      WHERE comercios.id = productos.comercio_id
        AND comercios.user_id = auth.uid()
    )
  )
  WITH CHECK (
    EXISTS (
      SELECT 1
      FROM comercios
      WHERE comercios.id = productos.comercio_id
        AND comercios.user_id = auth.uid()
    )
  );

COMMIT;

# Wrote hooks/use-pedido-desde-carrito.ts
import { useAuth } from '@/contexts/AuthContext';
import { useCart } from '@/contexts/CartContext';
import { supabase } from '@/constants/supabase';
import { useCallback, useState } from 'react';

export function usePedidoDesdeCarrito() {
  const { user } = useAuth();
  const { items, getSummary, clearCart } = useCart();
  const [loading, setLoading] = useState(false);
  const [error, setError] = useState<string | null>(null);

  const procesarPedido = useCallback(async () => {
    if (!user) {
      throw new Error('Debes iniciar sesión para procesar el pedido');
    }

    if (items.length === 0) {
      throw new Error('El carrito está vacío');
    }

    setLoading(true);
    setError(null);

    try {
      // 1. Obtener o crear comercio del usuario
      const { data: comercios, error: comerciosError } = await supabase
        .from('comercios')
        .select('id')
        .eq('user_id', user.id)
        .limit(1);

      if (comerciosError) throw comerciosError;

      let comercioId: number;

      if (comercios && comercios.length > 0) {
        comercioId = comercios[0].id;
      } else {
        // Crear comercio automáticamente
        const { data: nuevoComercio, error: crearComercioError } = await supabase
          .from('comercios')
          .insert([{ nombre: `Comercio de ${user.email?.split('@')[0] || 'Usuario'}` }])
          .select('id')
          .single();

        if (crearComercioError) throw crearComercioError;
        comercioId = nuevoComercio.id;
      }

      // 2. Verificar o crear cliente
      const { data: clientes, error: clientesError } = await supabase
        .from('clientes')
        .select('id')
        .eq('user_id', user.id)
        .limit(1);

      if (clientesError) throw clientesError;

      let clienteId: number;

      if (clientes && clientes.length > 0) {
        clienteId = clientes[0].id;
      } else {
        // Crear cliente automáticamente
        const { data: nuevoCliente, error: crearClienteError } = await supabase
          .from('clientes')
          .insert([{
            nombre: user.user_metadata?.username || user.email?.split('@')[0] || 'Cliente',
            telefono: null
          }])
          .select('id')
          .single();

        if (crearClienteError) throw crearClienteError;
        clienteId = nuevoCliente.id;
      }

      // 3. Generar pedido
      const summary = getSummary();
      const descripcion = items
        .map(item => `${item.product.nombre} x${item.cantidad}`)
        .join(', ');

      const { data: pedido, error: pedidoError } = await supabase
        .from('pedidos')
        .insert([{
          cliente_id: clienteId,
          comercio_id: comercioId,
          descripcion,
          monto: summary.total,
          estado: 1, // Pendiente
        }])
        .select('id')
        .single();

      if (pedidoError) throw pedidoError;

      // 4. Limpiar carrito
      clearCart();

      return pedido.id;
    } catch (err) {
      const msg = err instanceof Error ? err.message : 'Error al procesar pedido';
      setError(msg);
      throw err;
    } finally {
      setLoading(false);
    }
  }, [user, items, getSummary, clearCart]);

  return {
    procesarPedido,
    loading,
    error,
  };
}
